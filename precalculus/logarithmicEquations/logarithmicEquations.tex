\section{Basic Logarithmic Equations}
\begin{example}(Example 1)
Solve the equation $\log_2(x) = 6$.\\
We can rewrite this equation in exponential form as $2^6 = x$.
Thus, $x = 64$. 
\end{example}

\begin{problem}(Problem 1a)
Solve the equation $\log_5(x) = 3$.\\
$x = \answer{125}$
\end{problem}

\begin{problem}(Problem 1b)
Solve the equation $\log_{10}(x) = 4$.\\
$x = \answer{10000}$
\end{problem}

\begin{example}(Example 2)
Solve the equation $\log_3(x) = -3$.\\
    We can rewrite this equation in exponential form as $3^{-3} = x$.
    Thus, $x = \frac{1}{27} $.
\end{example}

\begin{problem}(Problem 2a)
Solve the equation $\log_2(x) = -4$.\\
$x = \answer{\frac{1}{16}}$
\end{problem}

\begin{problem}(Problem 2b)
Solve the equation $\log_9(x) = \frac{1}{2}$.\\
$x = \answer{3}$
\end{problem}
