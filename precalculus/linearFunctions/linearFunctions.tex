\section{Definition of Linear Functions}

A \textbf{linear} function is a function of the form $f(x) = ax + b$, where $a$ and $b$ are constants. 
If $a = 0$, the function is called a \textbf{constant} function. 
If $a \neq 0$, the function is called a non-constant linear function, or just a linear function. 

\begin{example}(Example 1)
Identify which of the following are linear functions:
\begin{enumerate}
\item $f(x) = 2x + 3$
\item $g(x) = x^2 + 1$
\item $h(x) = -5x$
\end{enumerate}
\begin{solution}
\begin{enumerate}
\item $f(x) = 2x + 3$ is a linear function since it is of the form $ax + b$ with $a = 2$ and $b = 3$.
\item $g(x) = x^2 + 1$ is not a linear function since it contains a term with $x^2$, which is not of the form $ax + b$.
\item $h(x) = -5x$ is a linear function since it is of the form $ax + b$ with $a = -5$ and $b = 0$.
\end{enumerate}
\end{solution}
\end{example}

\begin{example}(Example 2)
Classify the function as constant, linear or neither:
\begin{enumerate}
\item $f(x) = 4$ 
\item $g(x) = -3x + 2$ 
\item $h(x) = x^2 - 1$    
\end{enumerate} 
\begin{solution}
\begin{enumerate}
\item $f(x) = 4$ is a constant function since it is of the form $f(x) = b$ with $b = 4$.
\item $g(x) = -3x + 2$ is a linear function since it is of the form $f(x) = ax + b$ with $a = -3$ and $b = 2$.
\item $h(x) = x^2 - 1$ is neither a constant nor a linear function since it contains a term with $x^2$, which is not of the form $ax + b$ or $b$.    
\end{enumerate}
\end{solution}     
\end{example}

\section{Graphs of Linear Functions}

The graph of a linear function in the $(x,y)$-plane is a straight line.
To create the graph of a linear function, we can find two points on the line and then draw the unique line through those points.
\begin{example}(Example 3)
Graph the linear function $f(x) = 2x + 1$.\\
To graph the function, we can find two points on the line.
When $x = 0$, $f(0) = 2(0) + 1 = 1$, so the point $(0,1)$ is on the line.
When $x = 1$, $f(1) = 2(1) + 1 = 3$, so the point $(1,3)$ is on the line.
Now we can draw the line through the points $(0,1)$ and $(1,3)$ to graph the function.

\begin{tikzpicture}
\begin{axis}[
    axis lines = middle,
    xlabel = $x$,
    ylabel = $f(x)$,
    xmin=-1, xmax=3,
    ymin=-1, ymax=5,
    grid=both,
    width=10cm,
    height=10cm
]
\addplot[domain=-1:3, samples=100, color=blue]{2*x + 1};
\addplot[only marks, color=red] coordinates {(0,1) (1,3)};
\end{axis}
\end{tikzpicture}

\begin{image}
\begin{tikzpicture}
\draw (-1,0) -- (3,0) node[right] {$x$};

\filldraw[fill=red!30] (0,1) circle (3pt) node[below] {$(0,1)$} ;
\filldraw[fill=red!30] (1,3) circle (3pt) node[above] {$(1,3)$};
\end{tikzpicture}
\end{image}
\end{example}
