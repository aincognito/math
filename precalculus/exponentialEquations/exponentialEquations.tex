\section{Basic Exponential Equations}

\begin{example}(Example 1)
Solve the equation $2^x = 8$.\\
We can rewrite $8$ as $2^3$, so we have $2^x = 2^3$. 
Since the bases are the same, we can set the exponents equal to each other, giving us $x = 3$.
\end{example}

\begin{problem}(Problem 1a)
Solve the equation $2^x = 16$.\\
  $x = \answer{4}$
\end{problem}

\begin{problem}(Problem 1b)
Solve the equation $5^x = 25$.\\
  $x = \answer{2}$
\end{problem}

\begin{problem}(Problem 1c)
Solve the equation $10^x = 1000$.\\
$x = \answer{3}$
\end{problem}

\begin{example}(Example 2)
Solve the equation $3^{2x} = 27$.\\
    We can rewrite $27$ as $3^3$, so we have $3^{2x} = 3^3$. 
    Since the bases are the same, we can set the exponents equal to each other, giving us $2x = 3$. 
    Solving for $x$, we get $x = \frac{3}{2}$.
\end{example}

\begin{problem}(Problem 2a)
Solve the equation $4^{3x} = 64$.\\
$x = \answer{1}$
\end{problem}

\begin{problem}(Problem 2b)
Solve the equation $2^{5x} = 32$.\\
$x = \answer{1}$
\end{problem}

\begin{problem}(Problem 2c)
Solve the equation $3^{2x} = 81$.\\
$x = \answer{2}$
\end{problem}
